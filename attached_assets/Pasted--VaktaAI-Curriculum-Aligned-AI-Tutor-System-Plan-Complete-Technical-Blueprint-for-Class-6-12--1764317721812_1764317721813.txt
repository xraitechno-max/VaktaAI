```python
class Topic:
    """Individual topic/concept within a chapter"""
    topic_id: str                    # Unique identifier: "PHY_11_01_03" (Subject_Class_Chapter_Topic)
    topic_name: str                  # "Newton's Second Law"
    topic_name_hindi: str            # "न्यूटन का द्वितीय नियम"

    ncert_page_ref: str              # "NCERT Physics Class 11, Page 92-95"

    # Exam relevance
    board_weightage: float           # Marks in board exam (%)
    jee_weightage: float             # JEE Main weightage (%)
    jee_adv_weightage: float         # JEE Advanced weightage (%)
    neet_weightage: float            # NEET weightage (%)
    
    # Question patterns
    typical_question_types: List[str]  # ["Numerical", "Conceptual", "Application"]
    pyq_count_last_5_years: int        # Previous year questions


class Chapter:

    chapter_id: str                  # "PHY_11_01"
    chapter_name: str                # "Laws of Motion"
    chapter_name_hindi: str          # "गति के नियम"

    # NCERT mapping
    ncert_chapter_number: int        # 5
    ncert_book_name: str             # "Physics Part I"
    
    # Content structure
    topics: List[Topic]              # All topics in chapter
    total_periods: int               # School periods allocated
    
    # Exam metadata
    board_marks: int                 # Marks in board exam
    jee_questions_expected: int      # Expected questions in JEE
    neet_questions_expected: int     # Expected questions in NEET


class Subject:
```

### 2.2 Prerequisite Check Before Teaching

```python
def check_prerequisites_before_teaching(
    student_profile: StudentProfile,
    target_topic: str,
    curriculum_db: CurriculumDatabase
) -> PrerequisiteCheckResult:
    """
    Before teaching a topic, verify student has necessary background
    """
    
    topic = curriculum_db.get_topic(target_topic)
    prerequisites = topic.prerequisites
    
    result = PrerequisiteCheckResult()
    result.ready_to_learn = True
    result.missing_prerequisites = []
    result.weak_prerequisites = []
    result.suggested_review = []
    
    for prereq_id in prerequisites:
        mastery = student_profile.concept_mastery.get(prereq_id, 0.0)
        
        if mastery < 0.3:
            # Critical gap - must learn this first
            result.ready_to_learn = False
            result.missing_prerequisites.append(prereq_id)
        elif mastery < 0.6:
            # Weak foundation - brief review recommended
            result.weak_prerequisites.append(prereq_id)
            result.suggested_review.append(prereq_id)
    
    return result
```

```python
    def _get_exam_relevance(self, topic: Topic, exam: str) -> str:
        """Generate exam-specific relevance information"""
        
        if exam == "JEE_MAIN":
            return f"""

- Expected Questions: {topic.jee_questions_expected}
- Question Types: {', '.join(topic.jee_question_types)}
- Previous Year Frequency: {topic.pyq_count_last_5_years} questions in last 5 years
- Typical Difficulty: {topic.jee_difficulty_level}
- Time Allocation in Exam: {topic.recommended_time_mins} mins per question
"""
        elif exam == "NEET":
            return f"""
- NEET Weightage: {topic.neet_weightage}%
- Expected Questions: {topic.neet_questions_expected}
- Mostly: {topic.neet_question_type}  # "Conceptual" or "Application"
- NCERT Alignment: {topic.ncert_alignment_score}%
- Key NCERT Lines to Remember: {topic.ncert_key_lines}
"""
        elif exam == "BOARD":
            return f"""
- Board Marks: {topic.board_marks}
- Section in Paper: {topic.board_section}
- Question Format: {topic.board_question_format}
- Marking Scheme: {topic.marking_scheme}
"""
    
    def _get_prerequisite_status(self, student: StudentProfile, topic: Topic) -> str:
        """Check and report prerequisite mastery"""
        
        status_lines = []
        for prereq_id in topic.prerequisites:
            prereq = self.curriculum_db.get_topic(prereq_id)
            mastery = student.concept_mastery.get(prereq_id, 0.0)
            
            if mastery >= 0.7:
                status = "✅ Strong"
            elif mastery >= 0.4:
                status = "⚠️ Needs Review"
            else:
                status = "❌ Gap - Teach First"
            
            status_lines.append(f"- {prereq.topic_name}: {status} ({mastery*100:.0f}%)")
        
        return '\n'.join(status_lines)
    
    def _get_topic_specific_instructions(self, topic: Topic, student: StudentProfile) -> str:
        """Generate teaching instructions based on topic and student level"""
        
        instructions = []
        
        # Difficulty-based instructions
        if topic.difficulty_level >= 4:
            instructions.append("- This is a challenging topic. Use multiple examples and build up slowly.")
            instructions.append("- Break down into smaller sub-concepts.")
        
        # Exam-specific instructions
        if student.target_exam in ["JEE_MAIN", "JEE_ADVANCED"]:
            instructions.append(f"- Focus on problem-solving patterns for JEE.")
            instructions.append(f"- Cover shortcuts and time-saving techniques.")
            if topic.jee_numerical_heavy:
                instructions.append("- This topic is numerical-heavy in JEE. Practice calculations.")
        
        elif student.target_exam == "NEET":
            instructions.append("- Emphasize NCERT content and exact terminology.")
            instructions.append("- Focus on conceptual understanding over complex numericals.")
            if topic.requires_diagrams:
                instructions.append(f"- Key diagrams to cover: {', '.join(topic.key_diagrams)}")
        
        # Common mistakes
        if topic.common_misconceptions:
            instructions.append(f"- Watch for common mistakes: {', '.join(topic.common_misconceptions)}")
        
        return '\n'.join(instructions)
```

### 5.1 Coverage Tracking Schema

```python
class SyllabusCoverageTracker:
    """
    Track student's syllabus completion and mastery
    """
    
    def __init__(self, student_id: str, curriculum: CurriculumDatabase):
        self.student_id = student_id
        self.curriculum = curriculum
        self.coverage_data = {}
    
    def initialize_syllabus(self, student: StudentProfile):
        """Initialize tracking for student's syllabus"""
        
        subjects = self.curriculum.get_subjects_for_student(student)
        
        for subject in subjects:
            self.coverage_data[subject.subject_id] = {
                "total_chapters": subject.total_chapters,
                "chapters_started": 0,
                "chapters_completed": 0,
                "total_topics": sum(len(ch.topics) for ch in subject.chapters),
                "topics_mastered": 0,
                "chapter_details": {}
            }
            
            for chapter in subject.chapters:
                self.coverage_data[subject.subject_id]["chapter_details"][chapter.chapter_id] = {
                    "status": "not_started",  # not_started, in_progress, completed
                    "topics_total": len(chapter.topics),
                    "topics_seen": 0,
                    "topics_mastered": 0,
                    "mastery_score": 0.0,
                    "time_spent_mins": 0,
                    "last_studied": None,
                    "topic_details": {}
                }
                
                for topic in chapter.topics:
                    self.coverage_data[subject.subject_id]["chapter_details"][chapter.chapter_id]["topic_details"][topic.topic_id] = {
                        "status": "not_started",
                        "mastery": 0.0,
                        "attempts": 0,
                        "correct_attempts": 0,
                        "last_seen": None,
                        "next_review": None,  # Spaced repetition
                        "notes": []
                    }
    
    def update_topic_interaction(
        self, 
        topic_id: str, 
        interaction_type: str,  # "explanation", "practice", "doubt"
        performance: float,     # 0.0 to 1.0
        time_spent: int
    ):
        """Update tracking after student interacts with topic"""
        
        # Find topic in coverage data
        subject_id, chapter_id = self._get_parent_ids(topic_id)
        topic_data = self.coverage_data[subject_id]["chapter_details"][chapter_id]["topic_details"][topic_id]
        
        # Update status
        if topic_data["status"] == "not_started":
            topic_data["status"] = "learning"
            self.coverage_data[subject_id]["chapter_details"][chapter_id]["topics_seen"] += 1
        
        # Update mastery using weighted average
        old_mastery = topic_data["mastery"]
        topic_data["attempts"] += 1
        
        if interaction_type == "practice":
            # Practice performance directly affects mastery
            topic_data["mastery"] = 0.7 * old_mastery + 0.3 * performance
            if performance >= 0.8:
                topic_data["correct_attempts"] += 1
        elif interaction_type == "explanation":
            # Explanation gives smaller mastery boost
            topic_data["mastery"] = 0.9 * old_mastery + 0.1 * 0.5
        
        # Check if mastered
        if topic_data["mastery"] >= 0.7 and topic_data["correct_attempts"] >= 2:
            if topic_data["status"] != "mastered":
                topic_data["status"] = "mastered"
                self.coverage_data[subject_id]["chapter_details"][chapter_id]["topics_mastered"] += 1
                self.coverage_data[subject_id]["topics_mastered"] += 1
        
        # Update timestamps
        topic_data["last_seen"] = datetime.now()
        
        # Schedule next review (spaced repetition)
        topic_data["next_review"] = self._calculate_next_review(topic_data)
        
        # Update chapter status
        self._update_chapter_status(subject_id, chapter_id)
    
    def get_coverage_summary(self, student: StudentProfile) -> Dict:
        """Generate coverage summary for dashboard"""
        
        summary = {
            "overall_progress": 0.0,
            "subjects": {},
            "weak_areas": [],
            "strong_areas": [],
            "recommended_focus": [],
            "upcoming_reviews": []
        }
        
        total_topics = 0
        mastered_topics = 0
        
        for subject_id, subject_data in self.coverage_data.items():
            subject_progress = subject_data["topics_mastered"] / subject_data["total_topics"]
            total_topics += subject_data["total_topics"]
            mastered_topics += subject_data["topics_mastered"]
            
            summary["subjects"][subject_id] = {
                "progress": subject_progress,
                "chapters_completed": subject_data["chapters_completed"],
                "total_chapters": subject_data["total_chapters"],
                "topics_mastered": subject_data["topics_mastered"],
                "total_topics": subject_data["total_topics"]
            }
            
            # Identify weak and strong chapters
            for chapter_id, chapter_data in subject_data["chapter_details"].items():
                chapter = self.curriculum.get_chapter(chapter_id)
                
                if chapter_data["mastery_score"] < 0.4 and chapter_data["topics_seen"] > 0:
                    summary["weak_areas"].append({
                        "chapter": chapter.chapter_name,
                        "subject": subject_id,
                        "mastery": chapter_data["mastery_score"],
                        "exam_weightage": self._get_exam_weightage(chapter, student.target_exam)
                    })
                elif chapter_data["mastery_score"] >= 0.7:
                    summary["strong_areas"].append({
                        "chapter": chapter.chapter_name,
                        "subject": subject_id,
                        "mastery": chapter_data["mastery_score"]
                    })
        
        summary["overall_progress"] = mastered_topics / total_topics if total_topics > 0 else 0
        
        # Generate recommendations
        summary["recommended_focus"] = self._generate_focus_recommendations(summary, student)
        
        return summary
    
    def _generate_focus_recommendations(self, summary: Dict, student: StudentProfile) -> List[Dict]:
        """Generate prioritized study recommendations"""
        
        recommendations = []
        
        # Sort weak areas by exam weightage
        weak_sorted = sorted(
            summary["weak_areas"],
            key=lambda x: x["exam_weightage"],
            reverse=True
        )
        
        for weak_chapter in weak_sorted[:5]:  # Top 5 weak high-weightage chapters
            recommendations.append({
                "type": "weak_high_priority",
                "chapter": weak_chapter["chapter"],
                "reason": f"High exam weightage ({weak_chapter['exam_weightage']}%) but low mastery ({weak_chapter['mastery']*100:.0f}%)",
                "action": "Focus on strengthening fundamentals"
            })
        
        return recommendations
```

## 7. API ENDPOINTS FOR CURRICULUM SYSTEM

### 7.1 Core API Structure

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()

# Request Models
class TutoringRequest(BaseModel):
    student_id: str
    query: str
    topic_id: Optional[str] = None
    interaction_type: str  # "doubt", "explain", "practice", "revision"
    context: Optional[dict] = None

class PracticeRequest(BaseModel):
    student_id: str
    topic_id: str
    difficulty: Optional[str] = None  # "easy", "medium", "hard", "jee", "neet"
    count: int = 1

# Endpoints
@app.post("/tutor/interact")
async def tutor_interaction(request: TutoringRequest):
    """Main tutoring endpoint with curriculum awareness"""
    
    # 1. Load student profile
    student = await get_student_profile(request.student_id)
    
    # 2. Identify topic (if not provided)
    topic_id = request.topic_id or await identify_topic_from_query(
        request.query, 
        student
    )
    
    # 3. Generate curriculum context
    curriculum_context = await generate_curriculum_context(
        student=student,
        topic_id=topic_id,
        interaction_type=request.interaction_type
    )
    
    # 4. Check prerequisites
    prereq_status = await check_prerequisites(student, topic_id)
    
    # 5. Get class-level adaptations
    adaptations = get_class_adaptations(student)
    
    # 6. Generate AI response
    response = await generate_tutor_response(
        query=request.query,
        curriculum_context=curriculum_context,
        prereq_status=prereq_status,
        adaptations=adaptations,
        student=student
    )
    
    # 7. Update tracking
    await update_syllabus_coverage(
        student_id=request.student_id,
        topic_id=topic_id,
        interaction_type=request.interaction_type,
        performance=response.estimated_understanding
    )
    
    return {
        "response": response.text,
        "topic": topic_id,
        "prerequisites_checked": prereq_status,
        "coverage_update": response.coverage_delta
    }

@app.get("/student/{student_id}/syllabus")
async def get_syllabus_status(student_id: str):
    """Get student's complete syllabus with coverage status"""
    
    student = await get_student_profile(student_id)
    coverage = await get_coverage_tracker(student_id)
    
    syllabus = await get_curriculum_for_student(student)
    
    return {
        "student": student.dict(),
        "syllabus": syllabus,
        "coverage": coverage.get_coverage_summary(),
        "recommendations": coverage.get_recommendations()
    }

@app.get("/topic/{topic_id}/details")
async def get_topic_details(topic_id: str, student_id: str):
    """Get detailed topic information with student context"""
    
    student = await get_student_profile(student_id)
    topic = await curriculum_db.get_topic(topic_id)
    
    return {
        "topic": topic.dict(),
        "prerequisites": await get_prerequisites_with_status(topic_id, student),
        "exam_relevance": get_exam_relevance(topic, student.target_exam),
        "student_mastery": student.concept_mastery.get(topic_id, 0),
        "recommended_resources": get_resources(topic_id),
        "practice_available": await count_practice_questions(topic_id)
    }

@app.post("/practice/generate")
async def generate_practice(request: PracticeRequest):
    """Generate practice questions based on curriculum"""
    
    student = await get_student_profile(request.student_id)
    topic = await curriculum_db.get_topic(request.topic_id)
    
    # Determine difficulty
    if request.difficulty:
        difficulty = request.difficulty
    else:
        # Auto-adjust based on mastery
        mastery = student.concept_mastery.get(request.topic_id, 0)
        difficulty = calculate_optimal_difficulty(mastery)
    
    questions = await question_bank.get_questions(
        topic_id=request.topic_id,
        difficulty=difficulty,
        exam_type=student.target_exam,
        count=request.count,
        exclude=student.attempted_questions
    )
    
    return {
        "questions": questions,
        "difficulty": difficulty,
        "topic": topic.topic_name,
        "exam_pattern": get_exam_pattern_info(request.topic_id, student.target_exam)
    }
```
